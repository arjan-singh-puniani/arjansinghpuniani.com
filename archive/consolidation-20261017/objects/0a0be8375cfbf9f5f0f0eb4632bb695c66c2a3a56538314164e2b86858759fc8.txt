def die(msg: str) -> None:
    raise SystemExit(f"\nERROR: {msg}\n")

def find_repo(start: Path) -> Path:
    candidates = [start, *start.parents]
    for root in candidates:
        if (root / "rally-house-source-v16" / "package.json").exists() and (root / "arjan-portfolio" / "package.json").exists():
            return root
    die("Run this script from somewhere inside the arjansinghpuniani.com repository.")

ROOT = find_repo(Path.cwd().resolve())
SRC = ROOT / "rally-house-source-v16"
BACKUP = ROOT / ".rally-house-v17-backup"

targets = [
    SRC / "src/Game.ts",
    SRC / "src/world/World.ts",
    SRC / "src/ui/HUD.ts",
    SRC / "src/simulation/ObjectAffordances.ts",
    SRC / "src/tennis/RallySystem.ts",
    SRC / "src/core/AudioManager.ts",
    SRC / "src/persistence/migrations.ts",
    SRC / "index.html",
    SRC / "style.css",
]

if BACKUP.exists():
    die(f"{BACKUP} already exists. Remove it only if you intentionally want to re-apply this patch.")

for p in targets:
    if not p.exists():
        die(f"Missing expected file: {p.relative_to(ROOT)}")

# Guard against applying to an unknown baseline.
game_baseline = (SRC / "src/Game.ts").read_text()
world_baseline = (SRC / "src/world/World.ts").read_text()
if "const COSTS:Record<Placement['type'],number>={plant:30,bench:45,lamp:35,basket:25};" not in game_baseline:
    die("Game.ts no longer matches the inspected v16 baseline (COSTS marker missing).")
if "export interface Placement {id?:string;type:'plant'|'bench'|'lamp'|'basket';x:number;z:number}" not in world_baseline:
    die("World.ts no longer matches the inspected v16 baseline (Placement marker missing).")

BACKUP.mkdir(parents=True)
for p in targets:
    dst = BACKUP / p.relative_to(ROOT)
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(p, dst)

def write(path: Path, text: str) -> None:
    path.write_text(text)

def replace_once(text: str, old: str, new: str, label: str) -> str:
    count = text.count(old)
    if count != 1:
        die(f"{label}: expected exactly one match, found {count}. Your local source may have diverged.")
    return text.replace(old, new, 1)

def regex_once(text: str, pattern: str, repl: str, label: str) -> str:
    new, count = re.subn(pattern, repl, text, count=1, flags=re.S)
    if count != 1:
        die(f"{label}: expected exactly one regex match, found {count}. Your local source may have diverged.")
    return new

  private stagedBall(p:PendingShot){

  update(dt:number){
