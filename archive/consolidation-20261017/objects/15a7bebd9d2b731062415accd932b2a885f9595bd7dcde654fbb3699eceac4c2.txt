import assert from 'node:assert/strict';
import { chromium, browserOptions } from './runtime.mjs';

const baseUrl = process.env.RH_URL ?? 'http://127.0.0.1:8076/?debug';
const browser = await chromium.launch(browserOptions);
try {
  const page = await browser.newPage({ viewport: { width: 1280, height: 800 } });
  await page.goto(baseUrl, { waitUntil: 'domcontentloaded' });
  await page.waitForFunction(() => window.__rh?.ready === true, undefined, { timeout: 10000 });

  const result = await page.evaluate(() => {
    const world = window.__rh.game.world;
    const midday = world.lighting(12 * 60);
    const evening = world.lighting(19 * 60);
    const app = document.querySelector('#app');
    const clubOpacity = Number.parseFloat(getComputedStyle(app, '::after').opacity || '0');
    document.body.classList.add('championship-active');
    const matchOpacity = Number.parseFloat(getComputedStyle(app, '::after').opacity || '0');
    document.body.classList.remove('championship-active');
    return {
      clubOpacity,
      matchOpacity,
      middayPractical: midday.pointLights.slice(2).reduce((n, light) => n + light.intensity, 0),
      eveningPractical: evening.pointLights.slice(2).reduce((n, light) => n + light.intensity, 0),
      ambient: midday.ambient,
    };
  });

  assert.ok(result.clubOpacity >= 0.75, `club atmosphere should be visibly present: ${JSON.stringify(result)}`);
  assert.ok(result.matchOpacity <= 0.25, `Championship should suppress cozy overlay: ${JSON.stringify(result)}`);
  assert.ok(result.eveningPractical > result.middayPractical, `evening practicals should strengthen: ${JSON.stringify(result)}`);
  assert.ok(result.ambient < 0.9, `ambient field should preserve local-light contrast: ${JSON.stringify(result)}`);
  console.log('desktop: cozy atmosphere smoke PASS');
} finally {
  await browser.close();
}
