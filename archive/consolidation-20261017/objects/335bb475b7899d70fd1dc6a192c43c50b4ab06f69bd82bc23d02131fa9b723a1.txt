#!/usr/bin/env python3
from __future__ import annotations

from pathlib import Path
import shutil
import sys

ROOT = Path.cwd()
REPO = ROOT / 'arjan-portfolio'
PATCH_ROOT = Path(__file__).resolve().parent
PAYLOAD = PATCH_ROOT / 'payload' / 'arjan-portfolio'
BACKUP = ROOT / '.portfolio-rollup-backup-20260930'


def main() -> int:
    if not REPO.exists():
        print('Run this from the parent directory that contains ./arjan-portfolio', file=sys.stderr)
        return 1

    BACKUP.mkdir(exist_ok=True)
    for src in PAYLOAD.rglob('*'):
        if src.is_dir():
            continue
        rel = src.relative_to(PAYLOAD)
        dest = REPO / rel
        dest.parent.mkdir(parents=True, exist_ok=True)
        if dest.exists():
            backup = BACKUP / rel
            backup.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(dest, backup)
        shutil.copy2(src, dest)
        print(f'patched {rel}')

    print('\nPatch applied successfully.')
    print(f'Backups written to: {BACKUP}')
    print('Then run your normal install/build checks and start the dev server.')
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
