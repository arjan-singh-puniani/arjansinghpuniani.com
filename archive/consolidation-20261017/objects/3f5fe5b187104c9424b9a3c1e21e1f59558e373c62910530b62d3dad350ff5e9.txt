#!/usr/bin/env python3
from __future__ import annotations
import hashlib,json,shutil,sys
from datetime import datetime
from pathlib import Path

HERE=Path(__file__).resolve().parent
PAYLOAD=HERE/'payload'/'rally-house-source-v16'
BASELINE=json.loads((HERE/'BASELINE-SHA256.json').read_text())
TARGET=json.loads((HERE/'TARGET-SHA256.json').read_text())

def sha(path:Path)->str:
    return hashlib.sha256(path.read_bytes()).hexdigest()

def main():
    repo=Path(sys.argv[1] if len(sys.argv)>1 else '.').resolve()
    source=repo/'rally-house-source-v16'
    print('Rally House RC3.4 — cozy atmosphere installer')
    print('Repository:',repo)
    if not source.exists():
        raise SystemExit('ERROR: rally-house-source-v16 is missing. Restore RC3.2 first with Rally-House-Recovery-RC3.2, then rerun this installer.')

    print('Preflight: validating RC3.2/RC3.3 lineage...')
    changed=[]
    already=[]
    for rel,target_hash in TARGET.items():
        dst=source/rel
        if dst.exists() and sha(dst)==target_hash:
            already.append(rel); continue
        allowed=BASELINE.get(rel,[])
        if dst.exists():
            current=sha(dst)
            if current not in allowed:
                raise SystemExit(f'ERROR: preflight refused to overwrite modified file: {rel}\nCurrent SHA256: {current}\nExpected an RC3.2/RC3.3 lineage hash. No files were changed.')
        elif allowed:
            raise SystemExit(f'ERROR: expected baseline file is missing: {rel}. No files were changed.')
        changed.append(rel)

    if not changed:
        print('RC3.4 is already applied. No files changed.')
        return

    stamp=datetime.now().strftime('%Y%m%d-%H%M%S')
    backup=repo/f'rally-house-rc3.4-cozy-backup-{stamp}'
    backup.mkdir(parents=True,exist_ok=False)
    for rel in changed:
        dst=source/rel
        if dst.exists():
            out=backup/rel; out.parent.mkdir(parents=True,exist_ok=True); shutil.copy2(dst,out)

    try:
        for rel in changed:
            src=PAYLOAD/rel; dst=source/rel
            dst.parent.mkdir(parents=True,exist_ok=True); shutil.copy2(src,dst)
        for rel,target_hash in TARGET.items():
            dst=source/rel
            if not dst.exists() or sha(dst)!=target_hash:
                raise RuntimeError(f'post-write verification failed for {rel}')
    except Exception:
        for rel in changed:
            old=backup/rel; dst=source/rel
            if old.exists(): dst.parent.mkdir(parents=True,exist_ok=True); shutil.copy2(old,dst)
            elif dst.exists(): dst.unlink()
        raise

    print('Applied successfully.')
    print('Backup:',backup)
    print('Changed/added files:',len(changed))
    print('Version: 0.19.3-cozy.1')
    print('Next: python3', HERE/'release_candidate.py', repo)

if __name__=='__main__': main()
