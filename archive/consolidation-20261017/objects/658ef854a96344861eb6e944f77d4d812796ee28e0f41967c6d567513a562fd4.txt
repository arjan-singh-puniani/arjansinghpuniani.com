export function setupShoulderStudio({state,renderer,controls}) {
  const app=document.getElementById('app'), toolbar=document.getElementById('toolbar');
  const button=document.createElement('button'); button.type='button'; button.id='shoulder-studio-button'; button.textContent='Cuff Studio'; button.hidden=state.module!=='shoulder';
  toolbar.insertBefore(button,document.getElementById('section-button'));
  const panel=document.createElement('section'); panel.className='popover chrome shoulder-studio-panel'; panel.hidden=true;
  panel.innerHTML='<div class="popover-head"><strong>Rotator Cuff Studio</strong><span>Curated verified anatomy</span></div><div class="shoulder-studio-grid"></div><p class="shoulder-integrity">No tendon, labrum, bursa, capsule, ligament, or nerve is fabricated. Views use the verified meshes currently shipped with HoloAnatomy.</p>';
  app.append(panel); const grid=panel.querySelector('.shoulder-studio-grid');
  for(const view of Object.values(VIEWS)) { const b=document.createElement('button'); b.type='button'; b.innerHTML=`<strong>${view.label}</strong>`; b.onclick=()=>apply(view); grid.append(b); }
  button.onclick=()=>{panel.hidden=!panel.hidden};
  function apply(view){
    const vis=new Set(view.visible), ghost=new Set(view.ghost); state.structureOverrides={};
    for(const m of renderer.activeMeshes()) state.structureOverrides[m.part.id]=vis.has(m.part.id)?'visible':ghost.has(m.part.id)?'ghost':'hidden';
    state.selectedId=null; state.isolatedId=null; renderer.camera.yaw=view.yaw; renderer.camera.pitch=view.pitch; renderer.camera.distance=view.distance; renderer.camera.target=[0,0,0];
    controls.refreshSelection(); controls.renderLayers(); panel.hidden=true;
  }
  return {onModuleChanged(id){button.hidden=id!=='shoulder';if(id!=='shoulder')panel.hidden=true}};
}
