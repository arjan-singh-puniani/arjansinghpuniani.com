def die(msg: str) -> None:
    raise RuntimeError(msg)

def find_repo(start: Path) -> Path:
    for root in [start] + list(start.parents):
        src = root / "rally-house-source-v16" / "src"
        if (src / "Game.ts").exists() and (src / "entities" / "Character.ts").exists():
            return root
    raise RuntimeError("Run this from inside the Rally House repository/worktree.")

def backup_file(p: Path):
    dst = backup / p.relative_to(ROOT)
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(p, dst)

    backup_file(p)

def replace_once(text: str, old: str, new: str, label: str) -> str:
    n = text.count(old)
    if n != 1:
        die(f"{label}: expected exactly one match, found {n}")
    return text.replace(old, new, 1)

def regex_once(text: str, pattern: str, repl: str, label: str) -> str:
    out,n = re.subn(pattern,repl,text,count=1,flags=re.S)
    if n != 1:
        die(f"{label}: expected exactly one structural match, found {n}")
    return out

  assert.ok(restored.possessions.some(p=>p.owner==='steph'&&p.kind==='cup'));
