export function buildStructureOverrides(allIds, visible = [], ghost = [], hidden = []) {
  const overrides = Object.fromEntries(allIds.map(id => [id, 'ghost']));
  for (const id of visible) if (allIds.includes(id)) overrides[id] = 'visible';
  for (const id of ghost) if (allIds.includes(id)) overrides[id] = 'ghost';
  for (const id of hidden) if (allIds.includes(id)) overrides[id] = 'hidden';
  return overrides;
}

export function scoreRebuildDistance(distance) {
  const d = Math.max(0, Number(distance) || 0);
  if (d <= 0.18) return { grade: 'correct', label: 'Correct', complete: true };
  if (d <= 0.32) return { grade: 'close', label: 'Close', complete: false };
  return { grade: 'retry', label: 'Try again', complete: false };
}

export function chooseDifferentId(ids, previousId = null, rng = Math.random) {
  const pool = ids.filter(Boolean);
  if (!pool.length) return null;
  const alternatives = pool.filter(id => id !== previousId);
  const source = alternatives.length ? alternatives : pool;
  const value = Math.max(0, Math.min(0.999999, Number(rng()) || 0));
  return source[Math.floor(value * source.length)] || null;
}
