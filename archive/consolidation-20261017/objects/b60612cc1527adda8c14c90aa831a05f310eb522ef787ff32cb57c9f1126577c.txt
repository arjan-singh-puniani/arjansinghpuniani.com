def git_blob_sha(data: bytes) -> str:
    header = f"blob {len(data)}\0".encode()
    return hashlib.sha1(header + data).hexdigest()


def locate_app_root(candidate: Path) -> Path:
    candidate = candidate.resolve()
    if (candidate / "src/app/page.tsx").exists() and (candidate / "package.json").exists():
        return candidate
    if (candidate / "arjan-portfolio/src/app/page.tsx").exists():
        return candidate / "arjan-portfolio"
    raise SystemExit(
        "Could not locate arjan-portfolio. Run this from the repository root, "
        "the arjan-portfolio directory, or pass --repo /path/to/repository."
    )


def validate_projects(path: Path) -> str | None:
    data = path.read_bytes()
    actual = git_blob_sha(data)
    if actual == ORIGINAL["src/content/projects.ts"]:
        return None

    text = data.decode("utf-8")
    for variant in (PROJECT_GUTRUF, PROJECT_NEW):
        if text.count(variant) == 1 and text.count(PROJECT_OLD) == 0:
            normalized = text.replace(variant, PROJECT_OLD)
            if git_blob_sha(normalized.encode("utf-8")) == ORIGINAL["src/content/projects.ts"]:
                return None

    return (
        "changed baseline: src/content/projects.ts\n"
        f"  expected audited main or prior package variant\n  actual   {actual}"
    )


def validate_baseline(app_root: Path) -> list[str]:
    errors: list[str] = []
    for rel, expected in ORIGINAL.items():
        path = app_root / rel
        if not path.exists():
            errors.append(f"missing: {rel}")
            continue
        if rel == "src/content/projects.ts":
            err = validate_projects(path)
            if err:
                errors.append(err)
            continue
        actual = git_blob_sha(path.read_bytes())
        accepted = {expected}
        if rel in PREVIOUS:
            accepted.add(PREVIOUS[rel])
        if rel in GUTRUF_V1:
            accepted.add(GUTRUF_V1[rel])
        if rel in PLAIN_V1:
            accepted.add(PLAIN_V1[rel])
        if actual not in accepted:
            errors.append(
                f"changed baseline: {rel}\n"
                f"  expected audited main or prior package variant\n  actual   {actual}"
            )

    new_css = app_root / "src/app/home.module.css"
    if new_css.exists() and git_blob_sha(new_css.read_bytes()) not in {PREVIOUS_HOME_CSS, GUTRUF_V1_HOME_CSS, PLAIN_V1_HOME_CSS}:
        errors.append("src/app/home.module.css exists with content not created by a supported prior package")
    return errors


def copy_replacements(app_root: Path, backup_root: Path) -> None:
    package_root = Path(__file__).resolve().parent / "replacements/arjan-portfolio"
    for source in sorted(package_root.rglob("*")):
        if not source.is_file():
            continue
        rel = source.relative_to(package_root)
        target = app_root / rel
        if target.exists():
            backup = backup_root / rel
            backup.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(target, backup)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)


def patch_project_description(app_root: Path, backup_root: Path) -> None:
    path = app_root / "src/content/projects.ts"
    backup = backup_root / "src/content/projects.ts"
    backup.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(path, backup)
    text = path.read_text(encoding="utf-8")
    old_count = text.count(PROJECT_OLD)
    gutruf_count = text.count(PROJECT_GUTRUF)
    new_count = text.count(PROJECT_NEW)
    if old_count == 1 and gutruf_count == 0 and new_count == 0:
        text = text.replace(PROJECT_OLD, PROJECT_NEW)
        path.write_text(text, encoding="utf-8")
        return
    if old_count == 0 and gutruf_count == 1 and new_count == 0:
        text = text.replace(PROJECT_GUTRUF, PROJECT_NEW)
        path.write_text(text, encoding="utf-8")
        return
    if old_count == 0 and gutruf_count == 0 and new_count == 1:
        return
    raise RuntimeError(
        f"Unexpected ReasonOS description state: old={old_count}, prior={gutruf_count}, new={new_count}."
    )


def restore(app_root: Path) -> None:
    backup_root = app_root / BACKUP_NAME
    if not backup_root.exists():
        raise SystemExit("Backup folder does not exist; nothing to restore.")
    for source in sorted(backup_root.rglob("*")):
        if source.is_file():
            rel = source.relative_to(backup_root)
            target = app_root / rel
            target.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(source, target)
    css_backup = backup_root / "src/app/home.module.css"
    new_css = app_root / "src/app/home.module.css"
    if not css_backup.exists() and new_css.exists():
        new_css.unlink()
    print(f"Restored files from {BACKUP_NAME}")


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--repo", default=".", help="repository root or arjan-portfolio directory")
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--check", action="store_true", help="verify a supported baseline without writing")
    mode.add_argument("--apply", action="store_true", help="apply the upgrade after baseline verification")
    mode.add_argument("--restore", action="store_true", help="restore the pre-upgrade backup")
    args = parser.parse_args()

    app_root = locate_app_root(Path(args.repo))
    if args.restore:
        restore(app_root)
        return 0

    errors = validate_baseline(app_root)
    if errors:
        print("Baseline verification failed. No files were modified.", file=sys.stderr)
        for error in errors:
            print(f"- {error}", file=sys.stderr)
        return 2

    print(f"Baseline verified: {app_root}")
    print("Accepted audited GitHub main or a supported prior chat-generated upgrade.")
    if args.check:
        return 0

    backup_root = app_root / BACKUP_NAME
    if backup_root.exists():
        raise SystemExit(
            f"Backup already exists at {backup_root}. Restore/remove it before applying again."
        )
    backup_root.mkdir(parents=True)

    try:
        copy_replacements(app_root, backup_root)
        patch_project_description(app_root, backup_root)
    except Exception:
        for source in sorted(backup_root.rglob("*")):
            if source.is_file():
                rel = source.relative_to(backup_root)
                target = app_root / rel
                target.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(source, target)
        css_backup = backup_root / "src/app/home.module.css"
        new_css = app_root / "src/app/home.module.css"
        if not css_backup.exists() and new_css.exists():
            new_css.unlink()
        raise

    print("Final human-copy portfolio upgrade applied.")
    print(f"Backup: {BACKUP_NAME}")
    print("Required validation before deployment:")
    print("  npm run lint")
    print("  npm run typecheck")
    print("  npm run test")
    print("  npm run build")
    print("Then inspect /, /work, /notes, /about, and /contact on mobile and desktop.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
