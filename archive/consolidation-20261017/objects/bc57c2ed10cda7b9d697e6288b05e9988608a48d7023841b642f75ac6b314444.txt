import http from 'node:http';
import { readFile, stat } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const rootBase = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const root = path.resolve(rootBase, process.argv[2] || '.');
const publicDir = path.join(root, 'public');
const port = Number(process.env.PORT || 4173);
const mime = {
  '.html':'text/html; charset=utf-8', '.js':'text/javascript; charset=utf-8', '.css':'text/css; charset=utf-8',
  '.json':'application/json; charset=utf-8', '.obj':'text/plain; charset=utf-8', '.txt':'text/plain; charset=utf-8',
  '.png':'image/png', '.jpg':'image/jpeg', '.jpeg':'image/jpeg', '.webp':'image/webp', '.svg':'image/svg+xml'
};

async function resolveFile(rel) {
  const candidates = [path.resolve(root, rel), path.resolve(publicDir, rel)];
  for (let file of candidates) {
    if (!file.startsWith(root)) continue;
    try {
      if ((await stat(file)).isDirectory()) file = path.join(file, 'index.html');
      await stat(file);
      return file;
    } catch {
      // continue
    }
  }
  return null;
}

http.createServer(async (req,res)=>{
  try {
    const url = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
    const rel = decodeURIComponent(url.pathname).replace(/^\/+/, '') || 'index.html';
    const file = await resolveFile(rel);
    if (!file) throw new Error('Not found');
    const data = await readFile(file);
    const fromPublic = file.startsWith(publicDir + path.sep);
    res.writeHead(200, {
      'Content-Type': mime[path.extname(file).toLowerCase()] || 'application/octet-stream',
      'Cache-Control': fromPublic ? 'public, max-age=31536000, immutable' : 'no-cache'
    });
    res.end(data);
  } catch {
    res.writeHead(404, {'Content-Type':'text/plain; charset=utf-8'});
    res.end('Not found');
  }
}).listen(port,'127.0.0.1',()=>console.log(`HoloAnatomy: http://127.0.0.1:${port}`));
