#!/usr/bin/env python3
from pathlib import Path
import shutil, sys
from datetime import datetime

def die(msg):
    raise RuntimeError(msg)

def find_repo(start):
    for root in [start] + list(start.parents):
        game = root / "rally-house-source-v16" / "src" / "Game.ts"
        if game.exists():
            return root
    raise RuntimeError("Run this from inside the Rally House repository/worktree.")

ROOT = find_repo(Path.cwd().resolve())
GAME = ROOT / "rally-house-source-v16" / "src" / "Game.ts"
text = GAME.read_text()

required = [
    ("decor catalog", "DECOR_BY_ID,rotatedFootprint" in text),
    ("schema-8 era Game", "const LESSON_FEE=18;" in text),
    ("current court watch action", "{title:'Watch a rally',subtitle:'Coach Contessa ↔ Lucresia · rallies vary'" in text),
    ("current beginExhibition", "private beginExhibition()" in text),
]
missing = [name for name, ok in required if not ok]
if missing:
    die("Expected current Rally House v17.2 source. Missing: " + ", ".join(missing))

if "private playerWatchingRally=false;" in text and "private watchRallyAsPlayer()" in text:
    print("Rally House v17.3 Embodied Spectator is already installed.")
    sys.exit(0)

stamp = datetime.now().strftime("%Y%m%d-%H%M%S")
backup = ROOT / (".rally-house-v17-3-embodied-spectator-backup-" + stamp)
dst = backup / GAME.relative_to(ROOT)
dst.parent.mkdir(parents=True, exist_ok=True)
shutil.copy2(GAME, dst)

    count = text.count(old)
    if count != 1:
        die(f"{label}: expected exactly one match, found {count}")
    return text.replace(old, new, 1)
