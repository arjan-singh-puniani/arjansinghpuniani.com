import { cp, mkdir, readdir, rm, stat } from 'node:fs/promises';
import { existsSync } from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const dist = path.join(root, 'dist');
await rm(dist, { recursive:true, force:true });
await mkdir(dist, { recursive:true });
for (const item of ['index.html','styles.css','src']) {
  await cp(path.join(root,item), path.join(dist,item), { recursive:true });
}
const publicDir = path.join(root,'public');
if (existsSync(publicDir)) {
  for (const item of await readdir(publicDir)) {
    await cp(path.join(publicDir, item), path.join(dist, item), { recursive:true });
  }
}
const anatomyDir = path.join(root,'public','anatomy');
const assetsPresent = existsSync(anatomyDir) && (await readdir(anatomyDir)).some(n=>n.endsWith('.obj'));
console.log(`HoloAnatomy build complete: ${dist}`);
console.log(`Vendored anatomy: ${assetsPresent ? 'present' : 'not present (runtime falls back to pinned read-only mirror)'}`);
if (!assetsPresent) console.log('For a self-contained release run: python3 scripts/vendor_bodyparts3d.py && npm run build');
const info = await stat(path.join(dist,'index.html'));
console.log(`Entry document: ${info.size} bytes`);
